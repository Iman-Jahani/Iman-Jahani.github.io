// ---------- اضافه/آپدیت یک سلول ----------
function upsertCell(cell) {
  if (nodes.has(cell.id)) {
    // فقط محتوا را به‌روز کن، DOM را دست نزن
    const n = nodes.get(cell.id);
    if (!n.cm.hasFocus() && n.cm.getValue() !== (cell.code || '')) {
      n.cm.setValue(cell.code || '');
    }
    const newOut = cell.output || '';
    if (n.lastOutput !== newOut) {
      n.lastOutput = newOut;
      if (newOut) setOutput(n.outputEl, newOut,
        newOut.startsWith('Error') || newOut.startsWith('Pyodide'));
      else { n.outputEl.classList.remove('show'); n.outputEl.textContent = ''; }
    }
    // اگر order_index عوض شده، جابه‌جا کن
    n.el.dataset.order = String(cell.order_index ?? 0);
    return;
  }

  const n = createCellNode(cell);
  n.lastOutput = cell.output || '';
  nodes.set(cell.id, n);

  // درج در جای درست
  insertNodeInOrder(n);
}

// ---------- درج در جای درست ----------
function insertNodeInOrder(n) {
  const myOrder = Number(n.el.dataset.order) || 0;
  const all = [...cellsEl.children].filter(c => c.classList.contains('cell'));
  let before = null;
  for (const child of all) {
    const o = Number(child.dataset.order) || 0;
    if (o > myOrder) { before = child; break; }
  }
  if (before) cellsEl.insertBefore(n.el, before);
  else cellsEl.appendChild(n.el);
}

// ---------- حذف سلول ----------
function removeNode(id) {
  const n = nodes.get(id);
  if (!n) return;
  n.el.remove();
  nodes.delete(id);
}

// ---------- بارگذاری و رندر ----------
async function loadCells() {
  const { data, error } = await supabase
    .from('cells').select('*')
    .eq('session_id', SESSION_ID)
    .order('order_index', { ascending: true, nullsFirst: false })
    .order('created_at', { ascending: true });

  if (error) {
    toast('خطا: ' + error.message, 'err', 5000);
    return;
  }

  const incoming = data || [];

  // حذف سلول‌هایی که دیگر نیستند
  const incomingIds = new Set(incoming.map(c => c.id));
  for (const id of [...nodes.keys()]) {
    if (!incomingIds.has(id)) removeNode(id);
  }

  // اضافه/آپدیت
  for (const cell of incoming) upsertCell(cell);

  // حالت خالی
  const hasAny = nodes.size > 0;
  const emptyEl = cellsEl.querySelector('.empty');
  if (!hasAny && !emptyEl) {
    cellsEl.innerHTML = `
      

        
✨

        
هنوز سلولی نداری

        
روی «سلول جدید» بزن یا یک فایل ipynb آپلود کن.

      
`;
  } else if (hasAny && emptyEl) {
    emptyEl.remove();
  }

  document.getElementById('stats').textContent =
    nodes.size ? `${nodes.size} سلول · ${new Set([...nodes.values()].map(n => n.cell.author_name)).size} دانشجو` : '';
}

// ---------- افزودن سلول ----------
async function addCellAt(orderIndex) {
  const { error } = await supabase.from('cells').insert({
    session_id: SESSION_ID,
    author_id: user.id,
    author_name: displayName,
    code: '',
    output: '',
    order_index: orderIndex,
  });
  if (error) toast('خطا: ' + error.message, 'err', 5000);
}

// ---------- شروع ----------
async function init() {
  // ورود ناشناس
  let u = (await supabase.auth.getUser()).data.user;
  if (!u) {
    const { data, error } = await supabase.auth.signInAnonymously();
    if (error) { toast('خطای ورود: ' + error.message, 'err', 6000); return; }
    u = data.user;
  }
  user = u;

  displayName = localStorage.getItem('displayName');
  if (!displayName) {
    displayName = (prompt('اسمت را وارد کن:') || 'ناشناس').trim();
    localStorage.setItem('displayName', displayName);
  }
  document.getElementById('whoami').textContent = displayName;

  // لود اول
  cellsEl.innerHTML = `
در حال بارگذاری…
`;
  await loadCells();

  // دکمهٔ «سلول جدید»
  document.getElementById('addCell').onclick = async () => {
    const { data } = await supabase
      .from('cells').select('order_index')
      .eq('session_id', SESSION_ID)
      .order('order_index', { ascending: false })
      .limit(1);
    const max = Number(data?.[0]?.order_index) || 0;
    await addCellAt(max + 1);
  };

  // Realtime
  supabase
    .channel('cells-' + SESSION_ID)
    .on('postgres_changes', {
      event: '*', schema: 'public', table: 'cells',
      filter: `session_id=eq.${SESSION_ID}`,
    }, () => loadCells())
    .subscribe();

  // آپلود ipynb
  const fileInput = document.getElementById('fileInput');
  document.getElementById('uploadBtn').onclick = () => fileInput.click();

  fileInput.onchange = async (ev) => {
    const file = ev.target.files[0];
    if (!file) return;
    fileInput.value = '';

    try {
      const text = await file.text();
      let nb;
      try { nb = JSON.parse(text); }
      catch { return toast('فایل JSON معتبر نیست', 'err'); }
      if (!nb.cells || !Array.isArray(nb.cells))
        return toast('این فایل ipynb معتبر نیست', 'err');

      const codeCells = nb.cells.filter(c => c.cell_type === 'code');
      if (!codeCells.length) return toast('هیچ سلول کدی پیدا نشد', 'info');
      if (!confirm(`این فایل ${codeCells.length} سلول کد دارد. همه اضافه شوند؟`)) return;

      const { data: last } = await supabase
        .from('cells').select('order_index')
        .eq('session_id', SESSION_ID)
        .order('order_index', { ascending: false })
        .limit(1);
      const base = Number(last?.[0]?.order_index) || 0;

      const rows = codeCells.map((c, i) => ({
        session_id: SESSION_ID,
        author_id: user.id,
        author_name: displayName,
        code: Array.isArray(c.source) ? c.source.join('') : String(c.source || ''),
        output: '',
        order_index: base + i + 1,
      }));

      const btn = document.getElementById('uploadBtn');
      btn.disabled = true;
      toast(`در حال آپلود ${rows.length} سلول…`, 'info', 2000);

      const CHUNK = 20;
      for (let i = 0; i < rows.length; i += CHUNK) {
        const { error } = await supabase.from('cells').insert(rows.slice(i, i + CHUNK));
        if (error) {
          toast('خطای آپلود: ' + error.message, 'err', 6000);
          btn.disabled = false;
          return;
        }
      }
      btn.disabled = false;
      toast(`${rows.length} سلول اضافه شد`, 'ok');
    } catch (e) {
      console.error(e);
      toast('خطا در خواندن فایل: ' + e.message, 'err');
    }
  };

  // دانلود ipynb
  document.getElementById('exportBtn').onclick = async () => {
    toast('در حال ساخت فایل…', 'info', 1500);
    const { data, error } = await supabase
      .from('cells').select('*')
      .eq('session_id', SESSION_ID)
      .order('order_index', { ascending: true, nullsFirst: false })
      .order('created_at', { ascending: true });
    if (error) return toast('خطا: ' + error.message, 'err');

    const nb = {
      cells: data.map(c => ({
        cell_type: 'code',
        metadata: { author: c.author_name, created_at: c.created_at },
        source: (c.code || '').split('\n'),
        outputs: c.output ? [{
          output_type: 'stream', name: 'stdout',
          text: c.output.split('\n').map(l => l + '\n')
        }] : [],
        execution_count: null
      })),
      metadata: {
        kernelspec: { name: 'python3', display_name: 'Python 3' },
        language_info: { name: 'python', file_extension: '.py' }
      },
      nbformat: 4, nbformat_minor: 5
    };

    const blob = new Blob([JSON.stringify(nb, null, 1)], { type: 'application/json' });
    const a = document.createElement('a');
    a.href = URL.createObjectURL(blob);
    a.download = 'class-' + new Date().toISOString().slice(0, 10) + '.ipynb';
    a.click();
    URL.revokeObjectURL(a.href);
    toast('دانلود شد', 'ok');
  };
}

// ---------- اجرا ----------
init();
